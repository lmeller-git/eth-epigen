Preparing the data

dir.create("data")
## Warning in dir.create("data"): 'data' already exists
options(timeout = 3600)
download.file("https://ethz-ins.org/content/mouse_mm38_hippocampus.peakCounts.SE.rds",
  "data/hippocampus_peaks.rds",
  mode = "wb"
)

rds <- readRDS("data/hippocampus_peaks.rds")
sample_names <- colnames(rds)
rds$condition <- ifelse(grepl("FSS", sample_names), "FSS", "control")
rds$sex <- ifelse(grepl("female", sample_names), "female", "male")

rds$condition <- factor(rds$condition, levels = c("control", "FSS"))
rds$sex <- factor(rds$sex, levels = c("male", "female"))

rownames(rds) <- paste0("peak_", seq_len(nrow(rds)))

Exercise 1 - comparing stressed (denoted ‘FSS’ – forced swim stress) and control animals

Differential analysis and top 10 mos significant regions:

stress <- edgeR::DGEList(assay(rds), group = rds$condition)
stress <- edgeR::calcNormFactors(stress)
stress <- edgeR::estimateDisp(stress)
## Using classic mode.
stress_df <- as.data.frame(topTags(exactTest(stress), Inf))
top10_stress <- head(row.names(stress_df), 10)
cat("Top 10 most significant peaks for stress (control vs FSS):")
## Top 10 most significant peaks for stress (control vs FSS):
print(top10_stress)
##  [1] "peak_15757" "peak_14137" "peak_19358" "peak_8714"  "peak_2118" 
##  [6] "peak_4967"  "peak_869"   "peak_3430"  "peak_12264" "peak_17038"

Heatmap of normalized counts:

assay(rds, "logcpm_stress", withDimnames = FALSE) <- log1p(cpm(stress))
sechm(rds, top10_stress, assayName = "logcpm_stress", gaps_at = "condition")
